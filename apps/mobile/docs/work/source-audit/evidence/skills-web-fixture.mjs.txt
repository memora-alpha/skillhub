import http from 'node:http';
import {createRequire} from 'node:module';
const local=createRequire(process.cwd()+'/package.json');
const {ids,ownSkill,skillLevel,catalog,batch}=local('./tests/skills-fixtures.ts');
let records=[{...ownSkill}], mode='ok';
const profile={userId:ids.user,fullName:'Người kiểm thử Skills',headline:'Học và chia sẻ kỹ năng',bio:'Dữ liệu fixture cục bộ',timezone:'Asia/Ho_Chi_Minh',locale:'vi',avatarUrl:null,reputationScore:50,verifiedSkillCount:0,completedSessionCount:0};
const server=http.createServer(async(req,res)=>{
 res.setHeader('Access-Control-Allow-Origin','http://localhost:8088');res.setHeader('Access-Control-Allow-Headers','Content-Type,Authorization,Idempotency-Key');res.setHeader('Access-Control-Allow-Methods','GET,POST,PATCH,OPTIONS');
 if(req.method==='OPTIONS'){res.writeHead(204);res.end();return;}
 const u=new URL(req.url,'http://127.0.0.1:49152');const p=u.pathname.replace('/api/v1','');let raw='';for await(const c of req)raw+=c;const body=raw&&req.headers['content-type']?.includes('application/json')?JSON.parse(raw):{};
 const send=(data,status=200,page=false)=>{res.writeHead(status,{'Content-Type':'application/json'});res.end(JSON.stringify(status<400?{success:true,message:'OK',data,...(page?{meta:{pagination:{limit:20,hasMore:false,nextCursor:null}}}:{})}:{success:false,statusCode:status,message:data}));};
 if(u.pathname==='/_fixture'){if(req.method==='POST')mode=body.mode;return send({mode,count:records.length});}
 if(u.pathname==='/api/health')return send({status:'ok'});
 if(p==='/auth/login'||p==='/auth/refresh')return send({accessToken:'fixture-access',refreshToken:'fixture-refresh',tokenType:'Bearer',expiresIn:3600,refreshExpiresAt:'2035-01-01T00:00:00.000Z'});
 if(p==='/me')return send({userId:ids.user,email:'skills@example.test',emailVerified:true,accountStatus:'ACTIVE',roles:['USER'],profile});
 if(p==='/skills')return send([catalog],200,true);
 if(p===`/skills/${ids.skill}/levels`)return send([skillLevel]);
 if(p==='/me/skills'&&req.method==='GET'){if(mode==='denied')return send('SKILL_PERMISSION_REQUIRED',403);if(mode==='unavailable')return send('Unavailable',503);return send(records,200,true);}
 if(p==='/me/skills'&&req.method==='POST'){const row={...ownSkill,...body,id:crypto.randomUUID()};records.unshift(row);return send(row,201);}
 if(p.endsWith('/archive')){const id=p.split('/')[3];records=records.filter(r=>r.id!==id);return send({...ownSkill,id,version:2,archivedAt:new Date().toISOString()});}
 if(p.startsWith('/me/skills/')&&req.method==='PATCH'){const row=records.find(r=>r.id===p.split('/')[3]);if(!row)return send('NOT_FOUND',404);Object.assign(row,body,{version:row.version+1});return send(row);}
 if(p.endsWith('/evidence-files')&&req.method==='POST')return send({fileId:ids.file,state:'QUARANTINED'},202);
 if(p.includes('/evidence-files/'))return send({fileId:ids.file,state:'READY'});
 if(p.endsWith('/verifications')&&req.method==='POST')return send(batch,201);
 if(p.endsWith('/verifications'))return send([],200,true);
 if(p.includes('/verifications/'))return send(batch);
 if(p.startsWith('/profiles/')&&p.endsWith('/skills'))return send(records.filter(r=>r.isVisible),200,true);
 if(p.startsWith('/profiles/'))return send(profile);
 if(p.startsWith('/auth/logout'))return send(null);
 return send([],200,true);
});server.listen(49152,'127.0.0.1',()=>console.log('Skills-only fixture listening on 127.0.0.1:49152; no external API forwarding.'));
